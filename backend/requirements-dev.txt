# CI / local development tooling — pinned to exact versions.
#
# WHY: the CI workflow used to `pip install pytest pytest-cov ruff mongomock
# bandit pip-audit` with no versions, so every run resolved whatever was newest
# on PyPI that day. A tool's defaults changing then broke the build with no
# source change behind it (ruff's default rule set widened from E4/E7/E9/F to
# a much larger set, turning 2 lint errors into 280). Pinning here means a
# green run stays green until someone deliberately bumps a number.
#
# ruff and bandit are held to the same versions as .pre-commit-config.yaml so
# that the local hook and CI enforce an identical standard. Bump both together.

ruff==0.16.4          # == .pre-commit-config.yaml ruff-pre-commit rev v0.6.9
bandit==1.9.4       # == .pre-commit-config.yaml bandit rev 1.7.10
pip-audit==2.10.1
pytest==9.1.1
pytest-cov==7.1.0
mongomock==4.3.0
